# BGI定时一条龙
![1788366235794_d](/assets/img/inposts/bgi定时一条龙/1788366235794_d.png)
---
```
@echo off
color a
mode con: cols=80 lines=25
:: 关闭快速编辑
reg add "HKCU\Console" /v QuickEdit /t REG_DWORD /d 0 /f >nul 2>&1
chcp 65001 >nul

:: 开启CMD原生ANSI颜色支持（兼容Win10）
reg add HKCU\Console /v VirtualTerminalLevel /t REG_DWORD /d 1 /f >nul

set "RED=[91m"
set "GREEN=[92m"
set "YELLOW=[93m"
set "BLUE=[94m"
set "PURPLE=[95m"
set "CYAN=[96m"
set "WHITE=[97m"
set "RESET=[0m"

:: 配置
set "PATH_1_2=G:\20260707BGI"
set "PATH_3=F:\BetterGI Automation\betterGI0.55-"

cls
call :Echo "============================================================="
call :Echo "           BetterGI 批量执行脚本（无日志纯净版）"
call :Echo "============================================================="
call :Echo ""

call :Echo "[检测] 正在校验管理员权限..."
NET SESSION >nul 2>&1
if %errorlevel% neq 0 (
    call :Echo "[错误] 请右键以【管理员身份】运行此脚本！"
    pause >nul
    exit
)
call :Echo "[成功] 管理员权限校验通过！"
call :Echo ""

call :Echo "[清理] 正在检查并关闭残留进程（BetterGI/原神）..."
taskkill /f /im BetterGI.exe >nul 2>&1
taskkill /f /im yuanshen.exe >nul 2>&1
taskkill /f /im launch.exe >nul 2>&1
call :Echo "[成功] 已清理所有相关残留进程！"
call :Echo ""

:: 第一轮
call :Echo "[阶段1/3] 开始第一次执行 BetterGI"
cd /d "%PATH_1_2%" || (call :Echo "[错误] 目录切换失败！检查路径" & pause >nul & exit)
call :Echo "[启动] BetterGI.exe --startOneDragon"
.\BetterGI.exe --startOneDragon
call :Echo "[完成] 第一轮执行结束"
call :Echo ""

call :Echo "[等待] 等待10秒开启第二轮"
timeout /t 10 /nobreak >nul
call :Echo ""

:: 第二轮
call :Echo "[阶段2/3] 开始第二次执行 BetterGI"
cd /d "%PATH_1_2%" || (call :Echo "[错误] 目录切换失败！检查路径" & pause >nul & exit)
call :Echo "[启动] BetterGI.exe --startOneDragon"
.\BetterGI.exe --startOneDragon
call :Echo "[完成] 第二轮执行结束"
call :Echo ""

call :Echo "[等待] 等待10秒开启第三轮"
timeout /t 10 /nobreak >nul
call :Echo ""

:: 第三轮
call :Echo "[阶段3/3] 第三次执行（新路径）"
cd /d "%PATH_3%" || (call :Echo "[错误] 目录切换失败！检查路径" & pause >nul & exit)
call :Echo "[启动] BetterGI.exe --startOneDragon"
.\BetterGI.exe --startOneDragon
call :Echo "[完成] 第三轮执行结束"
call :Echo ""

call :Echo "============================================================="
call :Echo "[全部完成] 三轮任务执行完毕"
call :Echo "[提示] 3秒后自动关闭窗口"
call :Echo "============================================================="
timeout /t 3 /nobreak >nul
exit

:: 打印函数，bat原生获取时间，不再反复拉起PowerShell
:Echo
set "t=%time: =0%"
set "now=%date% %t:~0,8%"
echo %CYAN%[%now%] %~1%RESET%
goto :eof
```
